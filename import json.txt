import json
import re
from pathlib import Path
from collections import defaultdict

DIRECTORY_STRUCTURE_PATH = "directory_structure.json"
ALPHA_DOCS_PATH = "alpha_docs.json"


def normalize_name(name: str) -> str:
    name = name.replace("\\", "/")
    name = Path(name).name
    name = re.sub(r"\.(txt|md|docx|pdf)$", "", name, flags=re.IGNORECASE)
    name = re.sub(r"\s+", " ", name).strip()
    return name


def extract_base_and_version(filename: str):
    """
    Dosya adının sonundaki _1.0, _1.1, _2.0 gibi versiyonu yakalar.
    """
    clean = normalize_name(filename)

    match = re.search(r"_(\d+(?:\.\d+)+)$", clean)
    if match:
        version = match.group(1)
        base = clean[:match.start()]
        return base.strip(), version.strip()

    return clean.strip(), None


def walk_directory_structure(node, files=None):
    if files is None:
        files = []

    if node.get("type") == "file":
        name = node.get("name", "")
        base, version_from_name = extract_base_and_version(name)

        version = (
            node.get("source_version_folder")
            or version_from_name
        )

        files.append({
            "name": name,
            "base": base,
            "version": version,
            "path": node.get("path"),
            "extension": node.get("extension"),
            "size": node.get("size"),
        })

    for child in node.get("children", []):
        walk_directory_structure(child, files)

    return files


def load_alpha_docs(path):
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)

    file_names = data.get("file_names", [])

    parsed = []
    for name in file_names:
        base, version = extract_base_and_version(name)
        parsed.append({
            "name": name,
            "base": base,
            "version": version,
        })

    return parsed


with open(DIRECTORY_STRUCTURE_PATH, "r", encoding="utf-8") as f:
    directory_structure = json.load(f)

actual_files = walk_directory_structure(directory_structure)
alpha_files = load_alpha_docs(ALPHA_DOCS_PATH)


actual_map = defaultdict(set)
alpha_map = defaultdict(set)

for f in actual_files:
    if f["version"]:
        actual_map[f["base"]].add(f["version"])

for f in alpha_files:
    if f["version"]:
        alpha_map[f["base"]].add(f["version"])


missing_documents = []
missing_versions = []

for base, actual_versions in actual_map.items():
    alpha_versions = alpha_map.get(base)

    if alpha_versions is None:
        missing_documents.append({
            "document": base,
            "missing_versions": sorted(actual_versions),
        })
    else:
        diff = actual_versions - alpha_versions
        if diff:
            missing_versions.append({
                "document": base,
                "existing_in_alpha": sorted(alpha_versions),
                "missing_versions": sorted(diff),
            })


print("\n=== ALPHA_DOCS İÇİNDE TAMAMEN EKSİK DOKÜMANLAR ===")
for item in missing_documents:
    print(item)

print("\n=== ALPHA_DOCS İÇİNDE EKSİK VERSİYONLAR ===")
for item in missing_versions:
    print(item)


result = {
    "missing_documents": missing_documents,
    "missing_versions": missing_versions,
}

with open("alpha_missing_report.json", "w", encoding="utf-8") as f:
    json.dump(result, f, ensure_ascii=False, indent=2)

print("\nRapor oluşturuldu: alpha_missing_report.json")
